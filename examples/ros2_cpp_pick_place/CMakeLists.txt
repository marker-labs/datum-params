cmake_minimum_required(VERSION 3.16)
project(dparams_cpp_pick_place)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(rclcpp_action REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(trajectory_msgs REQUIRED)
find_package(control_msgs REQUIRED)
set(DPARAMS_HEADER_DIR "" CACHE PATH "Directory containing the freshly generated parameters.hpp")
if(NOT EXISTS "${DPARAMS_HEADER_DIR}/parameters.hpp")
  message(FATAL_ERROR "Set DPARAMS_HEADER_DIR to the directory containing generated parameters.hpp")
endif()
add_executable(pick_place src/pick_place.cpp)
target_compile_features(pick_place PRIVATE cxx_std_17)
target_include_directories(pick_place PRIVATE "${DPARAMS_HEADER_DIR}")
ament_target_dependencies(pick_place rclcpp rclcpp_action geometry_msgs sensor_msgs trajectory_msgs control_msgs)
add_executable(motor_sim src/motor_sim.cpp)
target_compile_features(motor_sim PRIVATE cxx_std_17)
ament_target_dependencies(motor_sim rclcpp rclcpp_action sensor_msgs trajectory_msgs control_msgs)
install(TARGETS pick_place motor_sim DESTINATION lib/${PROJECT_NAME})
ament_package()
